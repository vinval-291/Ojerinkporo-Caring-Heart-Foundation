# ===========================================================================
# public_html/.htaccess for ojerinkporofoundation.com
#
# Replaces whatever is there now. Paste the whole file.
#
# The block that matters for search is "old addresses" below. Every old
# WordPress URL currently answers 200 with the new homepage, so Google reads
# them as live pages and keeps them indexed with their old titles and copy.
# A 301 tells it the address moved; a 410 tells it the page is gone for good.
# Neither happens on its own.
# ===========================================================================

# --- one canonical address -------------------------------------------------
# The site's own canonical and og:url tags say non-www over https.
<IfModule mod_rewrite.c>
RewriteEngine On

RewriteCond %{HTTP_HOST} ^www\.(.+)$ [NC]
RewriteRule ^ https://%1%{REQUEST_URI} [L,R=301]

RewriteCond %{HTTPS} off
RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [L,R=301]
</IfModule>

# --- old addresses ---------------------------------------------------------
# Pages that have an equivalent on the new site: send the reader there and tell
# Google the address moved permanently.
<IfModule mod_rewrite.c>
RewriteRule ^home/?$                    / [L,R=301]
RewriteRule ^visionary/?$               /about/leadership [L,R=301]
RewriteRule ^donate/?$                  /support [L,R=301]
RewriteRule ^volunteer/?$               /partners [L,R=301]
RewriteRule ^faq/?$                     /contact [L,R=301]
RewriteRule ^events/grant/?$            /our-work/enterprise [L,R=301]
RewriteRule ^events(/.*)?$              /stories [L,R=301]
RewriteRule ^(our-)?programs?(/.*)?$    /our-work [L,R=301]
RewriteRule ^team/?$                    /about/leadership [L,R=301]
RewriteRule ^(blog|news)(/.*)?$         /stories [L,R=301]

# WordPress machinery and taxonomy archives. Nothing on the new site
# corresponds to these, so 410 Gone — Google drops a 410 far faster than a 404.
RewriteRule ^apus-event-category(/.*)?$   - [G,L]
RewriteRule ^apus-event-tag(/.*)?$        - [G,L]
RewriteRule ^(category|tag|author)(/.*)?$ - [G,L]
RewriteRule ^(feed|comments/feed)/?$      - [G,L]
RewriteRule ^wp-(content|includes|json)(/.*)?$ - [G,L]
RewriteRule ^(xmlrpc\.php|wp-login\.php|wp-cron\.php)$ - [G,L]
RewriteRule ^sitemap_index\.xml$          /sitemap.xml [L,R=301]
RewriteRule ^.*sitemap.*\.xml$            /sitemap.xml [L,R=301]
</IfModule>

# --- the site ---------------------------------------------------------------
<IfModule mod_rewrite.c>
RewriteBase /

# Leave the dashboard's server alone; the Node.js app answers there.
RewriteRule ^api(/|$) - [L]

# Serve real files and folders as they are.
RewriteCond %{REQUEST_FILENAME} -f [OR]
RewriteCond %{REQUEST_FILENAME} -d
RewriteRule ^ - [L]

# Every other address loads the site.
RewriteRule ^ /index.html [L]
</IfModule>

# --- caching ----------------------------------------------------------------
<IfModule mod_headers.c>
<FilesMatch "index\.html$">
Header set Cache-Control "no-cache"
</FilesMatch>
<FilesMatch "\.(js|css|jpe?g|png|webp|svg|woff2)$">
Header set Cache-Control "public, max-age=31536000, immutable"
</FilesMatch>
<FilesMatch "\.(xml|txt)$">
Header set Cache-Control "public, max-age=3600"
</FilesMatch>
</IfModule>

# There is deliberately no X-Robots-Tag here. That belongs on the preview,
# never on the live site.
